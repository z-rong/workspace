  - A-> B, B->C: onring, onanswer，B退出，挂断A、C

  initChat() {

      if (callInfo.isJoin == 1) {

      this.chatState = ChatState.ANSWERING

        this.hangUp(errorInfo.deviceId)
      }
      this.isChatting()
    })

      console.info('On answered:', deviceId);
      prompt.showToast({message:'对方已接听'})
      this.chatState = ChatState.CHATTING
      this.addDevice(deviceId)
      this.sufDevIdMap.set(this.selectSurfaceId, deviceId)
    })
    this.chat.on('noanswer', (deviceId) => {
      this.isChatting()
      prompt.showToast({message:'对方无应答'})
      console.info('On noanswer:', deviceId);
    })
    this.chat.on('reject', (deviceId) => {
      this.isChatting()
      prompt.showToast({message:'对方拒接'})
      console.info('On reject:', deviceId);
    })
    this.chat.on('busy', (deviceId) => {
      this.isChatting()
      prompt.showToast({message:'对方设备忙'})
      console.info('On busy:', deviceId);
    })

      console.info('On hangUp:', deviceId);
      prompt.showToast({message:'对方挂断'})
      this.removeDevice(deviceId)
      this.isChatting()
    })

      console.info('On join:', deviceId)
      this.chat.call(this.callType, deviceId, this.localSurfaceId, this.surfaceIdList[1][0]).then(() => {
        console.info('promise answer success.')

        console.error('promise answer failed, error code is :', err.code)

        this.isChatting()
      })

    this.chatState = ChatState.CALLING
    prompt.showToast({message:'呼叫中……'})
  }

  hangUp() {

            console.info('callback hangUp success', this.remoteDeviceId)
          }
          this.isChatting()
        })
      } else {
        this.chat.hangUp((err) => {

            console.info('callback hangUp all success')

          this.isChatting()
        })

        this.remoteDeviceId = this.getDeviceIdBySufId(this.selectSurfaceId)
//        if (this.remoteDeviceId == undefined) {
//          console
//          return

        console.info('promise hangUp remote device id = ', this.remoteDeviceId)
        this.chat.hangUp(this.remoteDeviceId).then(() => {
          console.info('promise hangUp success', this.remoteDeviceId)
          this.removeDevice(this.remoteDeviceId)
          this.isChatting()
        }).catch((err) => {
          console.error('promise hangUp failed, error code is :' + err.code)
          this.isChatting()
        })
        return

        console.info('promise hangUp all success')
        this.ChatList.length = 0
        this.isChatting()
      }).catch((err) => {
        console.error('promise hangUp all failed, error code is :' + err.code)

    })

  hangUp() {
    console.info('hangUp in')
    if (this.isCallback) {
      if (this.selectSurfaceId != 0) {
        this.remoteDeviceId = this.getDeviceIdBySufId(this.selectSurfaceId)
        this.chat.hangUp(this.remoteDeviceId, (err) => {
          if (err) {
            console.error('callback hangUp failed, error code is :' + err.code)
          } else {
            this.removeDevice(this.remoteDeviceId)
            console.info('callback hangUp success', this.remoteDeviceId)
          }
          this.isChatting()
        })
      } else {
        this.chat.hangUp((err) => {
          if (err) {
            console.error('callback hangUp all failed, error code is :' + err.code)
          } else {
            console.info('callback hangUp all success')
            this.ChatList.length = 0
          }
          this.isChatting()
        })
      }
    } else {
      if (this.selectSurfaceId != 0) {
        this.remoteDeviceId = this.getDeviceIdBySufId(this.selectSurfaceId)
        //        if (this.remoteDeviceId == undefined) {
        //          console
        //          return
        //        }
        console.info('promise hangUp remote device id = ', this.remoteDeviceId)
        this.chat.hangUp(this.remoteDeviceId).then(() => {
          console.info('promise hangUp success', this.remoteDeviceId)
          this.removeDevice(this.remoteDeviceId)
          this.isChatting()
        }).catch((err) => {
          console.error('promise hangUp failed, error code is :' + err.code)
          this.isChatting()
        })
        return
      }
      this.chat.hangUp().then(() => {
        console.info('promise hangUp all success')
        this.ChatList.length = 0
        this.isChatting()
      }).catch((err) => {
        console.error('promise hangUp all failed, error code is :' + err.code)
        this.isChatting()
      })
    }
  }
