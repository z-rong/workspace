    this.Init()

  findSurfaceId(deviceId: string) {
    for(let entry of this.surDevMap.entries()) {
      if (entry[1] == deviceId) {
        return entry[0]
      }
    }
  }

  removeXComponent(surfaceId: string) {
    let index = undefined
    for(let entry of this.controllers.entries()) {
      if(entry[1].getXComponentSurfaceId() == surfaceId){
        index = entry[0]
        break
      }
    }
    this.controllers.splice(index, 1)
  }

  Init() {
    console.log(' -- Init() enter');
    this.sinkClient = khSharingWfd.createSink()

    if (!this.sinkClient) {
      console.error(' -- DO NOT FOUND SERVICE!')
      return
    }

    this.sinkClient.on('error', (e) => {
      console.error('on error: ', JSON.stringify(e));
    });

    this.sinkClient.on('connectionChanged', (data) => {
      console.log('on connection changed data info, ', JSON.stringify(data))
      let surfaceId = undefined
      let connectionInfo = data
      switch(connectionInfo.state) {
        case 0:
          console.log('connection constructed')
          this.joinDeviceId = connectionInfo.macAddr
          console.log('join deviceId= ', this.joinDeviceId)
          let controller = new XComponentController()
          this.controllers.push(controller)
          break
        case 1:
          console.log('connection destroyed')
          surfaceId = this.findSurfaceId(data.macAddr)
          if (surfaceId == undefined) {
            console.error(' -- no surface is casting')
            return
          }
          this.surDevMap.delete(surfaceId)
          this.removeXComponent(surfaceId)
      }

    })
    console.log(' -- Init() out')
  }

  startPlay(deviceId: string, index: number) {
    let surfaceId = this.controllers[index].getXComponentSurfaceId()
    this.surDevMap.set(surfaceId, deviceId)
    console.log('on connection surfaceId=', surfaceId)
    this.setSurface(deviceId, surfaceId)
    this.joinDeviceId = undefined
  }

  start() {
    console.log(' -- start() enter ')
    if (this.sinkClient == undefined) {
      console.error('sinkClient is null')
      return
    }
    this.sinkClient.start((err) => {
      if (err == null) {
        console.info(' -- start() success.')
      } else {
        console.error(' -- start()  failed, error is ' + err.message)
      }
    })
    console.log(' -- start() out ')
  }

  stop() {
    console.log(' -- stop() enter ')
    if (this.sinkClient == undefined) {
      console.error('sinkClient is null')
      return
    }
    this.sinkClient.stop((err) => {
      if (err == null) {
        console.info(' -- stop() success.')
      } else {
        console.error(' -- stop()  failed, error is ' + err.message)
      }
    })

    console.log(' -- stop() out ')
  }
  setSurface(deviceId: string, surfaceId: string) {
    console.log(' -- SetSurface() enter');
    if (this.sinkClient == undefined) {
      console.error('sinkClient is null')
      return
    }
    if (deviceId == undefined || surfaceId == undefined) {
      console.error(' -- deviceId or surfaceId is undefined: ', deviceId, surfaceId)
      return
    }
    console.log('setSurface args: ', deviceId, surfaceId)
    this.sinkClient.setSurface(deviceId, surfaceId).then( () => {
      console.info(' -- setSurface() success.')
      let videoFormat = {codecType: 0, formatId: 4}
      let audioFormat = {codecType: 2, formatId: 6}
      this.setMediaFormat(deviceId, videoFormat, audioFormat)
    }).catch((err) =>{
      console.error(' -- setSurface() failed, error is ' + JSON.stringify(err))
    })
    console.log(' -- SetSurface() out')
  }


  setMediaFormat(deviceId: string, videoFormat: khSharingWfd.CodecAttr, audioFormat: khSharingWfd.CodecAttr) {
    console.log(' -- setMediaFormat enter')
    if (this.sinkClient == undefined) {
      console.error('sinkClient is null')
      return
    }
    if (deviceId == undefined || videoFormat == undefined || audioFormat == undefined){
      console.error(' -- setMediaFormat param is undefined')
      return
    }
    console.log('setMediaFormat() args', deviceId, JSON.stringify(videoFormat), JSON.stringify(audioFormat))
    this.sinkClient.setMediaFormat(deviceId, videoFormat, audioFormat, (err) => {
      if (err) {
        console.error(' -- setMediaFormat failed, err is: ', JSON.stringify(err))
      } else {
        console.log(' -- setMediaFormat success')
        this.play(deviceId)
      }
    })
    console.log(' -- setMediaFormat out')
  }

  play(deviceId: string) {
    console.log(' -- play() enter')
    if (deviceId == undefined) {
      console.error('deviceId is undefined')
      return
    }
    console.info(' -- play() args deviceId: ' + deviceId)
    this.sinkClient.play(deviceId, (err) => {
      if(err) {
        console.error(' -- play() failed, err: ', JSON.stringify(err))
      } else {
        console.log(' -- play() success')
      }
    })
    console.log(' -- play() out')
  }

  pause(deviceId: string) {
    console.log(' -- pause() enter')
    if (deviceId == undefined) {
      console.error('deviceId is undefined')
      return
    }
    console.info(' -- pause() args deviceId: ' + deviceId)
    this.sinkClient.pause(deviceId, (err) => {
      if(err) {
        console.error(' -- pause() failed, err: ', JSON.stringify(err))
      } else {
        console.log(' -- pause() success')
      }
    })
    console.log(' -- pause() out')

  mute(deviceId: string) {
    console.log(' -- mute() enter')
    if (deviceId == undefined) {
      console.error('mute is undefined')
      return
    }
    console.info(' -- mute() args deviceId: ' + deviceId)
    this.sinkClient.mute(deviceId, (err) => {
      if(err) {
        console.error(' -- mute() failed, err: ', JSON.stringify(err))
      } else {
        console.log(' -- mute() success')
      }
    })
    console.log(' -- mute() out')
  }

  unMute(deviceId: string) {
    console.log(' -- unMute() enter')
    if (deviceId == undefined) {
      console.error('unMute is undefined')
      return
    }
    console.info(' -- unMute() args deviceId: ' + deviceId)
    this.sinkClient.unMute(deviceId, (err) => {
      if(err) {
        console.error(' -- unMute() failed, err: ', JSON.stringify(err))
      } else {
        console.log(' -- unMute() success')
      }
    })
    console.log(' -- unMute() out')
  }

  setSceneType(deviceId: string, type: khSharingWfd.SceneType) {
    console.log(' -- setSceneType() enter')
    if (this.sinkClient == undefined) {
      console.error('sinkClient is null')
      return
    }
    if (deviceId == undefined || type == undefined){
      console.error(' -- setMediaFormat param is undefined')
      return
    }

      }
      else {
        console.log(' -- setSceneType() success');
      }
    })
    console.log('-- setSceneType out')
  }



  Release() {
    console.log(' -- release() enter')
    this.sinkClient.release().then(() => {
      delete this.sinkClient
      console.log(' -- release() success')
    }).catch((err) => {
      console.log(' -- release() err:', JSON.stringify(err))
    })
  }

  selectSurface(index: number) {
    if(!this.controllers[index]){
      return false
    }
    if(this.selectSurfaceId != this.controllers[index].getXComponentSurfaceId()) {
      return false
    }
    return true
  }

      return 1536
    }
    if (this.controllers.length == 2) {
      return 768
