```
#!/bin/bash

kmsg_file=$1
now=$(date +%Y%m%d%H%M%S)
save_dir=~/log/selinux_$now
if [ ! -d "$save_dir" ]; then
	mkdir $save_dir
fi

grep -rnw "permissive=0" $kmsg_file | awk -F "avc:" '{print $2}' | sort -u > $save_dir/${kmsg_file}.tmp
grep -rnw "Reap service" $kmsg_file | awk -F "Reap service" '{print $2}' | cut -d ',' -f 1 | tr -d ' ' | sort -u > $save_dir/failed_service.log
cd $save_dir
while read line; do
	scontext=$(echo $line |  awk -F "scontext="  '{print $2}' | cut -d ':' -f 3)
	tcontext=$(echo $line |  awk -F "tcontext="  '{print $2}' | cut -d ':' -f 3)
	tclass=$(echo $line |  awk -F "tclass="  '{print $2}' | cut -d ' ' -f 1)
	op=$(echo $line |  awk -F "{"  '{print $2}' | cut -d '}' -f 1)
    pid=$(echo $line |  awk -F "pid="  '{print $2}' | cut -d ' ' -f 1)
	if [ -f "${scontext}.te" ]; then
		is_op_exist=$(cat ${scontext}.te | grep -w "$tcontext:$tclass" | cut -d '{' -f 2 |  grep -w "$op")
		if [ -n "$is_op_exist" ]; then
			continue
		fi
		is_tclass_exist=$(cat ${scontext}.te | grep -nw "$tcontext:$tclass")
		if [ -n "$is_tclass_exist" ]; then
			line_num=$(echo $is_tclass_exist | cut -d ':' -f 1)
			sed -i "${line_num}s/}/$op }/" ${scontext}.te
            sed -i "s/  / /g" ${scontext}.te
            sed -i "${line_num} i #$line" ${scontext}.te
			continue
		fi
	fi
    echo "#${line}" >> ${scontext}.te 
	echo "allow $scontext $tcontext:$tclass { $op };" >> ${scontext}.te	
    sed -i "s/  / /g" ${scontext}.te
done <  ${kmsg_file}.tmp

rm ${kmsg_file}.tmp
```

打开selinux开关进行编译烧录，获取hilog和dmsg日志，脚本分析后获得各种te文件，找到分布式相关的，也就是d开头的，把它们复制粘贴到base/security/selinux/sepolicy/base相应文件中。然后重新编译这个库
