```400:428:foundation/multimedia/audio_framework/services/audio_policy/server/domain/device/src/pnp/audio_socket_thread.cpp
int32_t AudioSocketThread::AudioDetectUsbSoundCard(const AudioPnpUevent &audioPnpUevent)
{
    // 验证是否是USB音频设备的uevent
    CHECK_AND_RETURN_RET(strcmp(audioPnpUevent.subSystem, "sound") == 0, HDF_ERR_INVALID_PARAM);
    
    string devPath{audioPnpUevent.devPath};
    // 检查路径格式: /devices/.../usb5/.../sound/card3/pcmC3D0p
    const char *strs[] {"/usb", "/sound/card", "/pcmC"};
    size_t pos = 0;
    for (const char *str : strs) {
        pos = devPath.find(str, pos);
        CHECK_AND_RETURN_RET(pos != string::npos, HDF_ERR_INVALID_PARAM);
        pos += strlen(str);
    }
    
    AUDIO_INFO_LOG("devPath=%{public}s, action=%{public}s", 
                   audioPnpUevent.devPath, audioPnpUevent.action);
    
    // 提取card编号: C3D0 → cardNum=3
    auto cardNumStr = devPath.substr(pos, devPath.find('D', pos) - pos);
    
    if (strcmp(audioPnpUevent.action, "add") == 0) {
        AudioUsbManager::GetInstance().NotifySoundCardChange(cardNumStr, true);
    } else if (strcmp(audioPnpUevent.action, "remove") == 0) {
        AudioUsbManager::GetInstance().NotifySoundCardChange(cardNumStr, false);
    }
    
    return SUCCESS;
}
```

```414:436:foundation/multimedia/audio_framework/services/audio_policy/server/domain/device/src/pnp/audio_usb_manager.cpp
void AudioUsbManager::AddDeviceBySoundCard(uint32_t cardNum)
{
    // 等待/proc/asound/card3/usbbus文件就绪
    auto path = string{"/proc/asound/card"} + to_string(cardNum);
    WaitFile(path + "/" + "usbbus");
    
    // 读取声卡信息
    SoundCard card{.cardNum_ = cardNum};
    FillSoundCard(path, card);  // 读取usbbus, pcmC3D0p, pcmC3D0c
    
    // 从usbbus内容"5/6"解析出 busNum=5, devAddr=6
    auto usbAddr = GetUsbAddr(card.usbBus_);
    
    UsbAudioDevice dev {
        .usbAddr_ = usbAddr,
        .cardNum_ = card.cardNum_,
        .isCapturer_ = card.isCapturer_,  // 是否支持录音
        .isPlayer_ = card.isPlayer_,      // 是否支持播放
    };
    
    // ← 关键!调用USB服务获取设备名称
    UpdateDeviceName(usbAddr, dev.name_);
    
    audioDevices_.push_back(dev);
    NotifyDevice(dev, true);  // 通知上层
}
```

```438:454:foundation/multimedia/audio_framework/services/audio_policy/server/domain/device/src/pnp/audio_usb_manager.cpp
void AudioUsbManager::UpdateDeviceName(UsbAddr usbAddr, std::string &name)
{
    // 尝试1: 从缓存的pendingMap查找
    auto it = pendingMap_.find(usbAddr);
    if (it != pendingMap_.end()) {
        name = it->second;
        pendingMap_.erase(it);
        return;
    }
    
    // 尝试2: 调用USB服务获取所有USB音频设备
    vector<UsbAudioDevice> devs;
    GetUsbAudioDevices(devs);  // ← 这里调用UsbSrvClient::GetDevices()
    
    // 匹配busNum和devAddr,获取设备名称
    for (auto &item : devs) {
        if (item.usbAddr_ == usbAddr) {
            name = item.name_;  // "Logitech USB Headset H340"
            return;
        }
    }
    // 如果失败,name保持为空 → 日志显示name=-3
}
```

```484:537:foundation/multimedia/audio_framework/services/audio_policy/server/domain/device/src/pnp/audio_socket_thread.cpp
bool AudioSocketThread::AudioPnpUeventParse(const char *msg, const ssize_t strLength)
{
    struct AudioPnpUevent audioPnpUevent = {"", "", "", "", "", "", "", "", "", ""};
    
    // 解析uevent消息的key=value格式
    for (const char *msgTmp = msg; msgTmp < (msg + strLength);) {
        // 解析: ACTION=, DEVNAME=, SUBSYSTEM=, DEVPATH= 等字段
        const char *arrStrTmp[] = {
            UEVENT_ACTION, UEVENT_DEV_NAME, UEVENT_NAME, UEVENT_STATE,
            UEVENT_SUBSYSTEM, UEVENT_SWITCH_NAME, UEVENT_DEVPATH, ...
        };
        // ... 解析逻辑 ...
    }

    // 按subsystem分发到不同的处理函数
    if ((AudioAnalogHeadsetDetectDevice(&audioPnpUevent) == SUCCESS) ||
        (AudioHDMIDetectDevice(&audioPnpUevent) == SUCCESS) ||
        (AudioDpDetectDevice(&audioPnpUevent) == SUCCESS) ||
#ifdef USB_ENABLE
        (AudioDetectUsbSoundCard(audioPnpUevent) == SUCCESS) ||  // ← USB处理
#endif
        (AudioAnahsDetectDevice(&audioPnpUevent) == SUCCESS) ||
        (AudioMicBlockDevice(&audioPnpUevent) == SUCCESS)) {
        return true;
    }
    
    return false;
}
```

```334:380:foundation/multimedia/audio_framework/services/audio_policy/server/domain/device/src/device_status_listener.cpp
void DeviceStatusListener::OnPnpDeviceStatusChanged(const std::string &info)
{
    // 解析事件字符串: "EVENT_TYPE=1;DEVICE_TYPE=22;EVENT_NAME=...;DEVICE_ADDRESS=..."
    PnpDeviceType pnpDeviceType = PNP_DEVICE_UNKNOWN;
    PnpEventType pnpEventType = PNP_EVENT_UNKNOWN;
    
    if (sscanf_s(info.c_str(), "EVENT_TYPE=%d;DEVICE_TYPE=%d", 
                 &pnpEventType, &pnpDeviceType) < EVENT_PARAMS) {
        return;
    }
    
    // 转换为内部设备类型
    DeviceType internalDeviceType = GetInternalDeviceType(pnpDeviceType);
    // PNP_DEVICE_USB_HEADSET → DEVICE_TYPE_USB_HEADSET
    
    // 构造设备状态信息
    DStatusInfo statusInfo;
    statusInfo.deviceType = internalDeviceType;
    statusInfo.isConnected = (pnpEventType == PNP_EVENT_DEVICE_ADD);
    statusInfo.connectType = ConnectType::CONNECT_TYPE_LOCAL;
    
    // 通知设备观察者
    deviceObserver_.OnDeviceStatusUpdated(statusInfo);
}
```

### **🎯 关键发现**

1. **双线程监听机制**:
   - `OS_SocketEvent`: 监听Netlink uevent(硬件设备)
   - `OS_InputEvent`: 监听input事件(按键等)

2. **USB设备的两阶段识别**:
   - **阶段1**: Netlink uevent触发 → 识别为USB音频设备
   - **阶段2**: 调用USB服务 → 获取设备友好名称

3. **设备信息来源**:
   - **基本信息**: 从`/proc/asound/cardX/`读取(驱动提供)
   - **设备名称**: 从USB服务获取(读取USB描述符)

4. **为什么USB服务失败会影响设备加载**:
   - 虽然设备硬件被检测到,但**设备名称获取失败**
   - 导致`name=-3`,上层策略可能基于设备名称做判断
   - **实际音频功能可能正常**,只是缺少用户友好的描述信息

---

### **📌 总结**

音频设备上报流程是一个**事件驱动的多层架构**:

```
硬件插入 → 内核Uevent → Netlink Socket → AudioSocketThread解析
    → AudioUsbManager处理 → AudioPnpServer格式化 
    → DeviceStatusListener转换 → AudioCoreService加载
```

**USB耳机特殊之处**: 需要额外调用USB服务补充设备名称,这是当前问题的根源。解决方案应该是在USB服务不可用时**使用默认名称继续加载**,而不是阻塞整个设备初始化流程。
