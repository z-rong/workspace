#!/bin/bash


sleep_second=3
while [ true ]; do
    #press backhome key
    input -M 0 0 540 1020 && input -T -d 540 1020 && input -T -u 540 1020
    sleep 1

    #open camera app
    input -M 0 0 242 173 && input -T -d 242 173 && input -T -u 242 173
    sleep 1
   
    i=0
    while [ $i -lt 10 ]; do
        # 长按摄像头选择键
        input -T -d 1125 930 && sleep 3 && input -T -u 1125 930
        # select remote camera
        input -M 0 0 1170 502 && input -T -d 1170 502 && input -T -u 1170 502
        sleep 1
        # press ok
        input -M 0 0 960 724 && input -T -d 960 724 && input -T -u 960 724
        sleep 2

        # press select key    
        input -T -d 1125 930 && sleep 3 && input -T -u 1125 930
        # select local camera
        input -M 0 0 1164 420 && input -T -d 1164 420 && input -T -u 1164 420
        sleep 1
        input -M 0 0 960 724 && input -T -d 960 724 && input -T -u 960 724

        ((i++)) 
        echo ">>>>>>>>>>>>>>>>NO.$i"
    done

    echo "sleep $sleep_second ......"
    sleep $sleep_second 

    is_crash=$(find /data/log/faultlog -name "cppcrash-dhardware-*")
    if [ -n "$is_crash" ]; then
        echo ">>>>>>>>>> process crashed <<<<<<<<<<"
        break
    fi
done
