struct BaseDomainMsg : public BaseMsg {
    bool pullUpApp = false;
    std::string fromDevId;
    std::string toDevId;
    std::string fromRpcKey;
    std::string toRpcKey;
};

namespace Multicast {
    
struct MultiCastDomainReq : public BaseDomainMsg {
    SourceType sourceType;  //发送端源类型
	string canvasId;		//发送端源ID
};
    
//发送端主动拉流时，由Napi组织转发给接收端，接收端触发OnPull
struct PullReq : public MultiCastDomainReq {
	//默认直接拉第一个屏幕
};

//发送端AddDevice时，由Napi组织转发给接收端，接收端触发OnConnect
struct ConnectReq : public MultiCastDomainReq {
	vector<CodecAttr> videoAttrs; //支持的视频编码列表
};

//接收端在OnConnect回调中，由Napi组织转发给发送端，发送端触发OnAccept
struct AcceptReq : public MultiCastDomainReq {
	string ip;			//接收端ip
	uint16_t videoPort; //接收端视频收流端口
	uint16_t audioPort; //接收端音频收流端口
	CodecAttr videoAttr;	//接收端要求的视频格式
};

//发送端调用RemoveDevice 或 接收端调用UnBindDevice时，由Napi组织转发给对端，对端触发OnClose
struct CloseReq : public MultiCastDomainReq {
    int32_t clientType; //发起端类型：0:发送端， 1：接收端
};
    
//接收端调用EnableInputBackChannel由Napi组织转发给发送端native
struct ControlReq : public MultiCastDomainReq {
    bool enable; //反控起停标识
    CodecAttr videoAttr;
};

//拒绝对端投屏或拉起投屏
struct RejectReq : public MultiCastDomainReq {
    int32_t clientType; //发起端类型：0:发送端， 1：接收端
};

} //namespace Multicast

class Multicast {
public:
    Multicast();
    ~Multicast();

	/**通用**/
	//转发域间消息
	void ForwardDomainMsg(std::shared_ptr<BaseDomainMsg> &msg);
	//发现设备和拉起应用
	int32_t StartDiscover();
	int32_t StopDiscover();

	/**发送端**/
	//启用或关闭发送端服务(默认开启)
	int32_t EnableSource();
	int32_t DisableSource();
	//采集屏幕或窗口
	int32_t CreateCapture(CanvasType canvasType, uint64_t canvasId, CodecAttr videoAttr);
	int32_t CreateCaptureTmp(CanvasType canvasType, uint64_t canvasId, CodecAttr videoAttr, Rect rect);
	//删除屏幕采集
	int32_t DestroyCapture(uint64_t canvasId, CodecAttr videoAttr);
	//添加投屏，同一个分辨率的源不能两次投同一个设备
	int32_t AddDevice(string devId, uint64_t canvasId, CodecAttr videoAttr,RemoteAddr addr);
	//移除投屏
	int32_t RemoveDevice(string devId, uint64_t canvasId, CodecAttr videoAttr);
	//接受或关闭单路反控功能(被动调用)
	int32_t StartInput(string devId, uint64_t canvasId, CodecAttr videoAttr);
	int32_t StopInput(string devId, uint64_t canvasId, CodecAttr videoAttr);

	/**接收端**/
	int32_t EnableSink();
	int32_t DisableSink();
	//添加、删除投屏 (支持异显,预览开关和动作分开)
	int32_t BindDevice(string devId, uint64_t canvasId, CodecAttr videoAttr, RemoteAddr &recvAddr);
	int32_t UnBindDevice(string devId, uint64_t canvasId);
	//添加、删除接收端预览窗口,通过windowId获取surfaceId	
    int32_t AppendWindow(string devId, uint64_t canvasId, uint64_t windowId);
    int32_t RemoveWindow(string devId, uint64_t canvasId, uint64_t windowId);
    //暂时使用surfaceId
	int32_t AppendWindowTmp(string devId, uint64_t canvasId, uint64_t surfaceId, Rect rect);
    int32_t RemoveWindowTmp(string devId, uint64_t canvasId, uint64_t surfaceId);
    //方案一接口
    int32_t AppendWindowTmp(string devId, uint64_t canvasId, uint64_t surfaceId, uint64_t windowId,Rect rect);
    //方案二接口
    int32_t UpdateWindowRect(string devId, uint64_t canvasId, uint64_t surfaceId, Rect rect);

	//开始和停止预览
	int32_t Play(string devId, uint64_t canvasId);
	int32_t Close(string devId, uint64_t canvasId);
	//开启或关闭单路反控功能(只有主控端调用)
	int32_t StartInputBack(string devId, uint64_t canvasId);
	int32_t StopInputBack(string devId, uint64_t canvasId);

	int32_t SetListener(const std::shared_ptr<IMulticastListener> &listener);

private:
	struct RemoteAddr {
    	string devIp;
    	uint16_t videoPort;
    	uint16_t audioPort;
	};
	struct SourceKey {
		uint64_t sourceId;
    	int32_t formatId = -1;	//VIDEO_1080P_30
    	int32_t codecType = -1; //CODEC_H264、
	};

	struct SinkSession {
		uint32_t contextId;
		int32_t agentId;

    	string devIp;
    	uint16_t videoPort;
		uint16_t audioPort;
	};
	struct SourceSession {
        uint32_t contextId;
    	uint32_t agentId;
    	map<string,SinkSession> sinks; //key: devId(接收端设备id)
	};

	//发送端sources session存储
	map<SourceKey,SourceSession > sources_;
	//接收端session存储
	map<CastKey,SinkSession> sinks_;
};

## 关键流程
### 主投端投屏
#### 业务流程

1. 发送端APP—》发送端NAPI:AddDevice(devId, canvasId, sourceType,)—》
2. 发送端Native:ForwardDomainMsg(ConnectReq)—》RPC传输（附带拉起功能）—》
3. 接收端Navtive:OnInfo接收端NAPI::OnInfo(ConnectReq)—》协商判断编码类型并组织ConnectInfo—》
4. 接收端APP:OnConnect(CastConnectInfo)—》接收端NAPI:BindDevice(devId, canvasId, videoAttr)，接收端可同步进行NAPI:AppendWindow()和NAPI:Play()—》
5. 接收端Native:ForwardDomainMsg(AcceptReq)—》RPC传输-》
6. 发送端NAPI::OnInfo(AcceptReq)—》发送端APP:OnAccept(CastAcceptInfo castInfo)—》
7. 发送端NAPI调用Native:CreateCapture(devId, canvasId, sourceType, videoAttr)—》
8. 发送端NAPI调用Native:AddDevice(devId, canvasId, videoAttr, addr);

### 接收端拉流
#### 业务流程

1. 接收端APP—》接收端NAPI:PullCast(devId)—》
2. 接收端Native:ForwardDomainMsg(PullReq)—》RPC传输（附带拉起功能）—》
3. 发送端Navtive:OnInfo(PullReq)—》发送端NAPI:OnPull(CastInfo castInfo)—》
4. 发送端APP—》发送端NAPI:AddDevice(devId, canvasId, sourceType)，后续流程同“主投端投屏”

### 关闭投屏
#### 业务流程
（以发送端为例）

1. 发送端APP—》发送端NAPI:RemoveDevice—》发送端Native:ForwardDomainMsg(CloseReq)—》RPC传输—》
2. 发送端NAPI接着调用Native:RemoveDevice、Native:DestroyCapture进行资源释放
3. 接受端Navtive:OnInfo(CloseReq)—》发送端NAPI:OnClose(CastInfo castInfo)—》通知APP
4. 接受端NAPI接着调用Native:Close、Native:RemoveWindow、Native:UnBindDevice进行资源释放

### 反控开启
#### 业务流程

1. 接收端APP再调用NAPI:Play()过后，才能调用NAPI:StartInputBack()—》
2. 接受端Native:StartInputBack()—》接收端Scene:HandleAddInputBackCanvas()—》
3. 接收端Napi组织ControlReq消息—》接收端Native:ForwardDomainMsg(ControlReq)—》RPC传输—》
4. 发送端Navtive:OnInfo(ControlReq)—》发送端Scene:HandleAddInputCanvas()

### 主控端绑定显示窗口
#### 业务流程

1. 接收端APP调用XComponent:GetWindowId，获取当前XComponent组件所属窗口ID—》
2. 接收端APP调用Napi:appendWindow(windowId,surfaceId)—》接收端Native:AppendWindow()—》
3. 接收端Scene:HandleAppendWindow()记录对应的surfaceId和windowId—》
4. 接收端APP调用NAPI:StartInputBack()—》接受端Native:StartInputBack()—》
5. 接收端Scene:HandleAddInputBackCanvas(windowId,surfaceId)—》
6. 接收端InputBack:AddInputBackCanvas(windowId,surfaceId)—》
7. 接收端InputBack模块注册WindowsManager监听函数，窗口移动或变化时触发OnWindowUpdate() —》
8. 接收端InputBack模块在OnWindowUpdate(windowId，rect)中更新对应的Canvas的rect

### 主控端更新窗口区域
#### 业务流程

1. 接收端APP在XComponent位置和大小变更后，获取对应最新的Rect数据—》
2. 接收端APP调用Napi:updateWindowRect(surfaceId, rect, windowId, posX, posY)—》接收端Native:UpdateWindowRect()—》
3. 接收端Scene:HandleUpdateWindowRect()记录对应的surfaceId和最新rect—》
4. 接收端InputBack:UpdateCanvas(surfaceId，rect, windowId, posX, posY)中更新对应的Canvas的rect
